  // Older saved data stored glMeta/bankMeta as a single flat {fileName, count, ...}
  // object; this version stores a {files: [...]} list instead (to support "Add
  // rows"). Upgrade anything saved in the old shape so it doesn't crash the render.
  function normalizeMeta(meta){
    if (!meta) return null;
    if (Array.isArray(meta.files)) return meta;
    return { files: [{
      fileName: meta.fileName || 'imported file',
      sheetName: meta.sheetName || null,
      importedAt: meta.importedAt || new Date().toISOString(),
      skipped: meta.skipped || 0,
      skippedSamples: meta.skippedSamples || [],
      count: meta.count
    }] };
  }

  function loadState(){
    try {
      var raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) return defaultState();
      var parsed = JSON.parse(raw);
      var d = defaultState();
      var merged = Object.assign(d, parsed, { settings: Object.assign(d.settings, parsed.settings || {}) });
      merged.glMeta = normalizeMeta(merged.glMeta);
      merged.bankMeta = normalizeMeta(merged.bankMeta);
      return merged;
    } catch (e) {
      return defaultState();
    }
  }

  function saveState(){
    try { localStorage.setItem(STORAGE_KEY, JSON.stringify(state)); } catch (e) {}
  }

  function uid(){
    return 'x' + Date.now().toString(36) + Math.random().toString(36).slice(2, 8);
  }

  function fmtAmt(n){
    var sym = state.settings.currency || '';
    var neg = n < 0;
    var v = Math.abs(n).toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    return (neg ? '-' : '') + sym + v;
  }

  function fmtDate(iso){
    if (!iso) return '—';
    var d = new Date(iso + 'T00:00:00');
    if (isNaN(d)) return iso;
    return d.toLocaleDateString(undefined, { year: 'numeric', month: 'short', day: '2-digit' });
  }

  function daysBetween(iso1, iso2){
    var a = new Date(iso1 + 'T00:00:00');
    var b = new Date(iso2 + 'T00:00:00');
    if (isNaN(a) || isNaN(b)) return Infinity;
    return Math.round(Math.abs(a - b) / 86400000);
  }

  function toISODate(value){
    if (value instanceof Date && !isNaN(value)) {
      return value.getFullYear() + '-' + pad2(value.getMonth() + 1) + '-' + pad2(value.getDate());
    }
    if (typeof value === 'number') {
      var d = XLSX.SSF ? XLSX.SSF.parse_date_code(value) : null;
      if (d) return d.y + '-' + pad2(d.m) + '-' + pad2(d.d);
    }
    if (typeof value === 'string') {
      var s = value.trim();
      if (!s) return null;
      var m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})/);
      if (m) return m[1] + '-' + pad2(+m[2]) + '-' + pad2(+m[3]);
      m = s.match(/^(\d{1,2})[\/\-.](\d{1,2})[\/\-.](\d{2,4})$/);
      if (m) {
        var dd = +m[1], mm = +m[2], yy = +m[3];
        if (yy < 100) yy += 2000;
        if (dd > 12 && mm <= 12) { /* dd/mm/yyyy, keep */ }
        else if (mm > 12 && dd <= 12) { var t = dd; dd = mm; mm = t; }
        return yy + '-' + pad2(mm) + '-' + pad2(dd);
      }
      var parsed = new Date(s);
      if (!isNaN(parsed)) return toISODate(parsed);
    }
    return null;
  }

  function pad2(n){ return String(n).padStart(2, '0'); }

  function parseAmount(value){
    if (typeof value === 'number') return value;
    if (typeof value !== 'string') return NaN;
    var s = value.trim();
    if (!s) return NaN;
    var neg = false;
    if (/^\(.*\)$/.test(s)) { neg = true; s = s.slice(1, -1); }
    s = s.replace(/[^0-9.\-]/g, '');
    if (!s) return NaN;
    var n = parseFloat(s);
    if (isNaN(n)) return NaN;
    return neg ? -Math.abs(n) : n;
  }

  function isMatchedId(type, id){
    return state.matches.some(function(m){ return (type === 'gl' ? m.glIds : m.bankIds).indexOf(id) !== -1; });
  }

  function unmatchedOf(type){
    return state[type].filter(function(t){ return !isMatchedId(type, t.id); });
  }

  function sumOf(list){ return list.reduce(function(a, t){ return a + t.amount; }, 0); }

  function escapeHtml(s){
    return String(s == null ? '' : s).replace(/[&<>"']/g, function(c){
      return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
  }

  function pairKey(gId, bId){ return gId + '::' + bId; }

  function computeSuggestions(){
    var ug = unmatchedOf('gl'), ub = unmatchedOf('bank');
    var rejected = new Set(state.rejectedPairs);
    var candidates = [];
    for (var i = 0; i < ug.length; i++) {
      for (var j = 0; j < ub.length; j++) {
        var g = ug[i], b = ub[j];
        if (rejected.has(pairKey(g.id, b.id))) continue;
        var amtDiff = Math.abs(g.amount - b.amount);
        if (amtDiff > state.settings.amountTolerance) continue;
        var dDiff = daysBetween(g.date, b.date);
        if (dDiff > state.settings.dateToleranceDays) continue;
        var refMatch = g.reference && b.reference && g.reference.trim().toLowerCase() === b.reference.trim().toLowerCase();
        var confidence = (amtDiff < 0.005 && (dDiff === 0 || refMatch)) ? 'exact' : 'likely';
        var score = (refMatch ? -1000 : 0) + dDiff * 10 + amtDiff * 1000;
        candidates.push({ g: g, b: b, amtDiff: amtDiff, dDiff: dDiff, confidence: confidence, score: score });
      }
    }
    candidates.sort(function(a, b){ return a.score - b.score; });
    var usedG = new Set(), usedB = new Set(), out = [];
    candidates.forEach(function(c){
      if (usedG.has(c.g.id) || usedB.has(c.b.id)) return;
      usedG.add(c.g.id); usedB.add(c.b.id);
      out.push(c);
    });
    return out;
  }

  function handleFile(target, file, mode){
    var reader = new FileReader();
    reader.onload = function(e){
      try {
        var data = new Uint8Array(e.target.result);
        var wb = XLSX.read(data, { type: 'array', cellDates: true, raw: true });
        var sheetNames = wb.SheetNames.filter(function(n){
          var r = XLSX.utils.sheet_to_json(wb.Sheets[n], { header: 1, raw: true, defval: '' });
          return r.some(function(row){ return row.some(function(c){ return c !== '' && c != null; }); });
        });
        if (!sheetNames.length) { showToast('That file has no readable rows.', 'error'); return; }
        openMappingModal(target, wb, sheetNames, sheetNames[0], file.name, mode || 'replace');
      } catch (err) {
        showToast('Could not read that file: ' + err.message, 'error');
      }
    };
    reader.readAsArrayBuffer(file);
  }

  function guessColumn(headers, keywords){
    for (var i = 0; i < headers.length; i++) {
      var h = String(headers[i]).toLowerCase();
      for (var k = 0; k < keywords.length; k++) {
        var kw = keywords[k];
        // Very short abbreviations ("dr", "cr") need a whole-word match — otherwise
        // "cr" matches inside unrelated words like "Description".
        if (kw.length <= 2) {
          var re = new RegExp('\\b' + kw + '\\b');
          if (re.test(h)) return i;
        } else if (h.indexOf(kw) !== -1) {
          return i;
        }
      }
    }
    return -1;
  }

  function rowsForSheet(wb, sheetName){
    var rows = XLSX.utils.sheet_to_json(wb.Sheets[sheetName], { header: 1, raw: true, defval: '' });
    return rows.filter(function(r){ return r.some(function(c){ return c !== '' && c != null; }); });
  }

  function openMappingModal(target, wb, sheetNames, sheetName, fileName, mode){
    pendingImport = { target: target, wb: wb, sheetNames: sheetNames, fileName: fileName, mode: mode || 'replace' };
    loadSheet(sheetName);
  }

  function loadSheet(sheetName){
    var rows = rowsForSheet(pendingImport.wb, sheetName);
    var headerRow = rows[0].map(function(c, i){ return c !== '' ? String(c) : 'Column ' + (i + 1); });
    var colCount = rows[0].length;

    var guesses = {
      date: guessColumn(headerRow, ['date']),
      description: guessColumn(headerRow, ['desc', 'narrat', 'particular', 'detail', 'memo']),
      reference: guessColumn(headerRow, ['ref', 'cheque', 'check', 'trans']),
      amount: guessColumn(headerRow, ['amount', 'value']),
      debit: guessColumn(headerRow, ['debit', 'dr']),
      credit: guessColumn(headerRow, ['credit', 'cr'])
    };
    var amountMode = (guesses.debit !== -1 && guesses.credit !== -1 && guesses.amount === -1) ? 'split' : 'single';

    pendingImport.sheetName = sheetName;
    pendingImport.rows = rows;
    pendingImport.hasHeader = true;
    pendingImport.colCount = colCount;
    pendingImport.amountMode = amountMode;
    pendingImport.map = guesses;
    // Debit/Credit mean opposite things on a bank statement vs a GL: a bank debit is
    // money leaving your account (the bank owes you less), while a GL debit to a bank/
    // cash account is money coming in (the asset increases). Default each side to its
    // normal convention; the toggle below lets the user override it either way.
    pendingImport.debitIsOutflow = pendingImport.target === 'bank';
    renderModal();
  }

  function colOptions(selectedIdx){
    var hr = pendingImport.hasHeader ? pendingImport.rows[0] : null;
    var opts = '<option value="-1">— none —</option>';
    for (var i = 0; i < pendingImport.colCount; i++) {
      var label = hr ? (hr[i] !== '' ? String(hr[i]) : 'Column ' + (i + 1)) : 'Column ' + (i + 1);
      opts += '<option value="' + i + '"' + (i === selectedIdx ? ' selected' : '') + '>' + escapeHtml(label) + '</option>';
    }
    return opts;
  }

  function getSelVal(id){ var el = document.getElementById(id); return el ? parseInt(el.value, 10) : -1; }

  function confirmImport(){
    var p = pendingImport;
    var dateCol = getSelVal('map-date');
    var descCol = getSelVal('map-description');
    var refCol = getSelVal('map-reference');
    var amountCol = p.amountMode === 'single' ? getSelVal('map-amount') : -1;
    var debitCol = p.amountMode === 'split' ? getSelVal('map-debit') : -1;
    var creditCol = p.amountMode === 'split' ? getSelVal('map-credit') : -1;

    if (dateCol === -1 || (p.amountMode === 'single' && amountCol === -1) || (p.amountMode === 'split' && debitCol === -1 && creditCol === -1)) {
      document.getElementById('import-warning').innerHTML = '<div class="warn-banner">Please choose at least a date column and an amount column before importing.</div>';
      return;
    }

    var dataRows = p.rows.slice(p.hasHeader ? 1 : 0);
    var headerOffset = p.hasHeader ? 2 : 1; // human-readable row number, 1-based, accounting for the header row
    var out = [], skippedSamples = [];
    dataRows.forEach(function(r, idx){
      var iso = toISODate(r[dateCol]);
      var amt;
      if (p.amountMode === 'single') {
        amt = parseAmount(r[amountCol]);
      } else {
        var dv = debitCol !== -1 ? parseAmount(r[debitCol]) : 0;
        var cv = creditCol !== -1 ? parseAmount(r[creditCol]) : 0;
        if (isNaN(dv)) dv = 0;
        if (isNaN(cv)) cv = 0;
        // Debit/Credit are mirrored between a bank statement and a GL bank/cash account —
        // p.debitIsOutflow (set from the "What does Debit mean here?" choice) picks which way.
        amt = p.debitIsOutflow ? (cv - dv) : (dv - cv);
      }
      if (!iso || isNaN(amt)) {
        var reasonParts = [];
        if (!iso) reasonParts.push('unreadable date (' + JSON.stringify(r[dateCol]) + ')');
        if (isNaN(amt)) reasonParts.push('unreadable amount');
        skippedSamples.push({ row: idx + headerOffset, reason: reasonParts.join(', '), preview: r.slice(0, Math.min(r.length, 5)).join(' | ') });
        return;
      }
      out.push({
        id: uid(),
        date: iso,
        description: descCol !== -1 ? String(r[descCol] != null ? r[descCol] : '').trim() : '',
        reference: refCol !== -1 ? String(r[refCol] != null ? r[refCol] : '').trim() : '',
        amount: Math.round(amt * 100) / 100
      });
    });

    if (!out.length) {
      document.getElementById('import-warning').innerHTML = '<div class="warn-banner">No valid rows found — check the date and amount columns and try again.</div>';
      return;
    }

    if (p.mode === 'append') {
      var dupIdx = computeDuplicates(p.target, out);
      if (dupIdx.length) {
        var dupSet = {};
        dupIdx.forEach(function(i){ dupSet[i] = true; });
        var keepCount = out.length - dupIdx.length;
        document.getElementById('import-warning').innerHTML =
          '<div class="warn-banner">' +
            '<div>' + dupIdx.length + ' of these ' + out.length + ' row(s) look identical to ones already loaded (same date, amount and description) — probably already reconciled.</div>' +
            '<div style="margin-top:8px;display:flex;gap:8px;flex-wrap:wrap;">' +
              '<button class="btn-primary btn-sm" id="dup-skip" type="button">Skip duplicates, add the other ' + keepCount + '</button>' +
              '<button class="btn-ghost btn-sm" id="dup-all" type="button">Add all ' + out.length + ' anyway</button>' +
            '</div>' +
          '</div>';
        document.getElementById('dup-skip').onclick = function(){ finalizeImport(out.filter(function(_, i){ return !dupSet[i]; }), skippedSamples); };
        document.getElementById('dup-all').onclick = function(){ finalizeImport(out, skippedSamples); };
        return;
      }
    }

    finalizeImport(out, skippedSamples);
  }

  function txSignature(t){
    return t.date + '|' + t.amount.toFixed(2) + '|' + (t.description || '').trim().toLowerCase();
  }

  function computeDuplicates(target, rows){
    var existing = {};
    state[target].forEach(function(t){ existing[txSignature(t)] = true; });
    var seenInBatch = {};
    var dupIdx = [];
    rows.forEach(function(r, i){
      var sig = txSignature(r);
      if (existing[sig] || seenInBatch[sig]) dupIdx.push(i);
      seenInBatch[sig] = true;
    });
    return dupIdx;
  }

  function finalizeImport(finalOut, skippedSamples){
    var p = pendingImport;
    var fileRecord = {
      fileName: p.fileName,
      sheetName: p.sheetNames.length > 1 ? p.sheetName : null,
      importedAt: new Date().toISOString(),
      skipped: skippedSamples.length,
      skippedSamples: skippedSamples.slice(0, 30),
      count: finalOut.length
    };

    if (p.mode === 'append' && state[p.target + 'Meta']) {
      state[p.target] = state[p.target].concat(finalOut);
      state[p.target + 'Meta'].files.push(fileRecord);
    } else {
      state[p.target] = finalOut;
      state[p.target + 'Meta'] = { files: [fileRecord] };
      // full replace: drop matches/selections referencing the side just replaced
      state.matches = state.matches.filter(function(m){
        return p.target === 'gl' ? m.glIds.every(function(id){ return finalOut.some(function(t){ return t.id === id; }); }) || m.glIds.length === 0
                                  : m.bankIds.every(function(id){ return finalOut.some(function(t){ return t.id === id; }); }) || m.bankIds.length === 0;
      });
      selection.gl.clear(); selection.bank.clear();
    }

    saveState();
    closeModal();
    renderAll();
    var verb = p.mode === 'append' ? 'Added ' : 'Imported ';
    if (skippedSamples.length) {
      showToast(verb + finalOut.length + ' rows — ' + skippedSamples.length + ' row(s) skipped (date or amount didn\'t parse). See "view" next to the file for details.', 'info');
    } else {
      showToast(verb + finalOut.length + ' rows from ' + p.fileName + '.', 'info');
    }
  }

  function showSkippedModal(target){

    var totalSkipped = 0;

      (f.skippedSamples || []).forEach(function(s){ rows.push(Object.assign({ fileName: f.fileName }, s)); });

          '<div class="modal-body">' +
            '<p style="margin:0 0 12px;font-size:12.5px;color:var(--ink-dim);">These rows couldn\'t be read as transactions and were left out. Re-check the column mapping (or the source file) if that\'s more than expected — spreadsheet row numbers below include the header row.</p>' +

  function csvField(v){
    var s = v == null ? '' : String(v);
    if (/[",\n\r]/.test(s)) s = '"' + s.replace(/"/g, '""') + '"';
    return s;
  }

  function csvRow(fields){ return fields.map(csvField).join(',') + '\r\n'; }

  function buildReportCsv(){
    var glTotal = sumOf(state.gl), bankTotal = sumOf(state.bank);
    var variance = Math.round((glTotal - bankTotal) * 100) / 100;
    var ug = unmatchedOf('gl'), ub = unmatchedOf('bank');
    var out = '';

    out += csvRow(['Ledger Match — Reconciliation Report']);
    out += csvRow(['Generated', new Date().toLocaleString()]);
    out += csvRow(['GL file(s)', state.glMeta ? state.glMeta.files.map(function(f){ return f.fileName; }).join('; ') : '—']);
    out += csvRow(['Bank file(s)', state.bankMeta ? state.bankMeta.files.map(function(f){ return f.fileName; }).join('; ') : '—']);
    out += '\r\n';

    out += csvRow(['SUMMARY']);
    out += csvRow(['GL total', fmtAmt(glTotal)]);
    out += csvRow(['Bank total', fmtAmt(bankTotal)]);
    out += csvRow(['Variance (GL − Bank)', fmtAmt(variance)]);
    out += csvRow(['Matched pairs', state.matches.length]);
    out += csvRow(['Unmatched GL lines', ug.length]);
    out += csvRow(['Unmatched bank lines', ub.length]);
    out += '\r\n';

    out += csvRow(['MATCHED']);
    out += csvRow(['Match #', 'Side', 'Date', 'Description', 'Reference', 'Amount', 'Match method', 'Match variance']);
    state.matches.forEach(function(m, i){
      var glItems = state.gl.filter(function(t){ return m.glIds.indexOf(t.id) !== -1; });
      var bankItems = state.bank.filter(function(t){ return m.bankIds.indexOf(t.id) !== -1; });
      var variance = m.confidence === 'variance' ? fmtAmt(m.variance) : fmtAmt(0);
      glItems.forEach(function(t, j){
        out += csvRow([i + 1, 'GL', fmtDate(t.date), t.description, t.reference, fmtAmt(t.amount), j === 0 ? m.method + ' / ' + m.confidence : '', j === 0 ? variance : '']);
      });
      bankItems.forEach(function(t){
        out += csvRow([i + 1, 'Bank', fmtDate(t.date), t.description, t.reference, fmtAmt(t.amount), '', '']);
      });
    });
    if (!state.matches.length) out += csvRow(['(none)']);
    out += '\r\n';

    out += csvRow(['UNMATCHED — GENERAL LEDGER']);
    out += csvRow(['Date', 'Description', 'Reference', 'Amount']);
    ug.forEach(function(t){ out += csvRow([fmtDate(t.date), t.description, t.reference, fmtAmt(t.amount)]); });
    if (!ug.length) out += csvRow(['(none)']);
    out += '\r\n';

    out += csvRow(['UNMATCHED — BANK STATEMENT']);
    out += csvRow(['Date', 'Description', 'Reference', 'Amount']);
    ub.forEach(function(t){ out += csvRow([fmtDate(t.date), t.description, t.reference, fmtAmt(t.amount)]); });
    if (!ub.length) out += csvRow(['(none)']);

    return out;
  }

  var downloadsCap; // undefined = not yet resolved, null = unavailable, object = ready

  function getDownloadsCap(){
    if (downloadsCap !== undefined) return Promise.resolve(downloadsCap);
    if (typeof window.claude === 'undefined' || !window.claude.use) {
      downloadsCap = null;
      return Promise.resolve(null);
    }
    return window.claude.use('downloads').then(function(cap){ downloadsCap = cap; return cap; }).catch(function(){ downloadsCap = null; return null; });
  }

  function downloadCsvViaBrowser(filename, csv){
    try {
      var blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
      var url = URL.createObjectURL(blob);
      var a = document.createElement('a');
      a.href = url;
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      setTimeout(function(){ URL.revokeObjectURL(url); }, 1000);
      showToast('Report downloaded.', 'info');
    } catch (err) {
      showToast('Could not download the report' + (err && err.message ? ': ' + err.message : '.'), 'error');
    }
  }

  function exportReport(){
    if (!state.gl.length && !state.bank.length) {
      showToast('Upload a GL or bank file first — there\'s nothing to report on yet.', 'error');
      return;
    }
    var stamp = new Date().toISOString().slice(0, 10);
    var filename = 'ledger-match-report-' + stamp + '.csv';
    var csv = buildReportCsv();
    getDownloadsCap().then(function(downloads){
      if (downloads) {
        return downloads.save({ filename: filename, data: csv }).then(function(){ showToast('Report saved.', 'info'); });
      }
      // Plain browser download — used whenever this page isn't running inside
      // the claude.ai artifact viewer (e.g. hosted on GitHub Pages).
      downloadCsvViaBrowser(filename, csv);
    }).catch(function(err){
      if (err && err.code === 'declined') return;
      if (err && err.code === 'rate_limited') { showToast('A save prompt is already open — try again in a moment.', 'error'); return; }
      downloadCsvViaBrowser(filename, csv);
    });
  }

  function acceptSuggestion(gId, bId, confidence){
    state.matches.push({ id: uid(), glIds: [gId], bankIds: [bId], method: 'auto', confidence: confidence, createdAt: new Date().toISOString() });
    saveState();
    renderAll();
  }

  function rejectSuggestion(gId, bId){
    state.rejectedPairs.push(pairKey(gId, bId));
    saveState();
    renderAll();
  }

  function acceptAllExact(){
    var sugg = computeSuggestions().filter(function(s){ return s.confidence === 'exact'; });
    sugg.forEach(function(s){
      state.matches.push({ id: uid(), glIds: [s.g.id], bankIds: [s.b.id], method: 'auto', confidence: 'exact', createdAt: new Date().toISOString() });
    });
    saveState();
    renderAll();
  }

  function unmatch(matchId){
    state.matches = state.matches.filter(function(m){ return m.id !== matchId; });
    saveState();
    renderAll();
  }

  function matchSelected(){
    var glIds = Array.from(selection.gl), bankIds = Array.from(selection.bank);
    if (!glIds.length || !bankIds.length) return;
    var glSum = sumOf(state.gl.filter(function(t){ return glIds.indexOf(t.id) !== -1; }));
    var bankSum = sumOf(state.bank.filter(function(t){ return bankIds.indexOf(t.id) !== -1; }));
    var variance = Math.round((glSum - bankSum) * 100) / 100;
    state.matches.push({ id: uid(), glIds: glIds, bankIds: bankIds, method: 'manual', confidence: Math.abs(variance) <= state.settings.amountTolerance ? 'exact' : 'variance', variance: variance, createdAt: new Date().toISOString() });
    selection.gl.clear(); selection.bank.clear();
    saveState();
    renderAll();
  }

  function toggleSelect(type, id){
    var set = selection[type];
    if (set.has(id)) set.delete(id); else set.add(id);
    renderUnmatchedGrid();
    renderSelectBar();
  }

        });

        var fileInputAdd = document.getElementById('file-add-' + target);
        fileInputAdd.onchange = function(e){ if (e.target.files[0]) handleFile(target, e.target.files[0], 'append'); fileInputAdd.value = ''; };

      body.innerHTML = '<div class="empty-state"><div class="es-icon">✓</div>Nothing to suggest right now — upload both sides or check back after matching more.</div>';

  function renderUnmatchedGrid(){
    var ug = unmatchedOf('gl'), ub = unmatchedOf('bank');
    var el = document.getElementById('unmatched-grid');

    // Rebuilding innerHTML wipes the scroll position of the scrollable table
    // bodies below — save it per side and restore it after, so ticking a
    // checkbox near the bottom of a long list doesn't jump back to the top.
    var prevBodies = el.querySelectorAll('table.tx tbody');
    var scrollTops = { gl: prevBodies[0] ? prevBodies[0].scrollTop : 0, bank: prevBodies[1] ? prevBodies[1].scrollTop : 0 };

    el.innerHTML =
      '<div><div class="tx-col-head"><span class="col-title gl">General ledger</span><span class="col-sum">' + fmtAmt(sumOf(ug)) + '</span></div>' + txTable('gl', ug) + '</div>' +
      '<div><div class="tx-col-head"><span class="col-title bank">Bank statement</span><span class="col-sum">' + fmtAmt(sumOf(ub)) + '</span></div>' + txTable('bank', ub) + '</div>';

    var newBodies = el.querySelectorAll('table.tx tbody');
    if (newBodies[0]) newBodies[0].scrollTop = scrollTops.gl;
    if (newBodies[1]) newBodies[1].scrollTop = scrollTops.bank;

    el.querySelectorAll('[data-check]').forEach(function(cb){
      cb.addEventListener('change', function(){
        var id = cb.getAttribute('data-check');
        var type = state.gl.some(function(t){ return t.id === id; }) ? 'gl' : 'bank';
        toggleSelect(type, id);
      });
    });
  }

  function renderSelectBar(){
    var bar = document.getElementById('select-bar');
    if (!selection.gl.size && !selection.bank.size) { bar.style.display = 'none'; return; }
    bar.style.display = 'flex';
    var glSum = sumOf(state.gl.filter(function(t){ return selection.gl.has(t.id); }));
    var bankSum = sumOf(state.bank.filter(function(t){ return selection.bank.has(t.id); }));
    var variance = Math.round((glSum - bankSum) * 100) / 100;
    document.getElementById('sel-gl-sum').textContent = fmtAmt(glSum);
    document.getElementById('sel-bank-sum').textContent = fmtAmt(bankSum);
    var vEl = document.getElementById('sel-variance');
    vEl.textContent = fmtAmt(variance);
    vEl.className = Math.abs(variance) <= state.settings.amountTolerance ? 'variance-ok' : 'variance-bad';
    document.getElementById('btn-match-selected').disabled = !selection.gl.size || !selection.bank.size;
  }

      body.innerHTML = '<div class="empty-state">No matches confirmed yet.</div>';

      var bankItems = state.bank.filter(function(t){ return m.bankIds.indexOf(t.id) !== -1; });
      var glDesc = glItems.map(function(t){ return fmtDate(t.date) + ' · ' + escapeHtml(t.description || '—'); }).join('<br>');
      var bankDesc = bankItems.map(function(t){ return fmtDate(t.date) + ' · ' + escapeHtml(t.description || '—'); }).join('<br>');

      return '<div class="matched-row">' +

    });
  }

  // Each panel renders independently — a bug in one (or unexpected shape in the
  // saved data feeding it) surfaces as one broken section plus a toast, instead
  // of a single uncaught error blanking the entire page.
  function renderAll(){
    var failures = [];
    function safe(fn, label){
      try { fn(); } catch (err) { console.error('Ledger Match render error (' + label + '):', err); failures.push(label); }
    }
    safe(renderSummary, 'summary');
    safe(renderIntake, 'data sources');
    safe(renderSuggestions, 'suggested matches');
    safe(renderUnmatchedGrid, 'unmatched transactions');
    safe(renderSelectBar, 'selection bar');
    safe(renderMatched, 'matched list');
    if (failures.length) {
      showToast('Some sections couldn\'t render: ' + failures.join(', ') + '. If this persists, try Clear all data.', 'error');
    }
  }

  // ---------------------------------------------------------------------
  // static wiring
  // ---------------------------------------------------------------------
  // Wrapped defensively: if the saved state is ever in a shape a future version
  // doesn't expect, this reports it instead of leaving a silently blank page.
  try {
    document.getElementById('btn-reset-all').onclick = clearAll;
    document.getElementById('btn-export').onclick = exportReport;
    document.getElementById('btn-accept-all').onclick = acceptAllExact;
    document.getElementById('btn-match-selected').onclick = matchSelected;
    document.getElementById('btn-settings').onclick = function(){
      var panel = document.getElementById('settings-panel');
      panel.style.display = panel.style.display === 'none' ? 'block' : 'none';
    };
    document.getElementById('set-date-tol').onchange = function(e){
      state.settings.dateToleranceDays = Math.max(0, parseInt(e.target.value, 10) || 0);
      saveState(); renderAll();
    };
    document.getElementById('set-amt-tol').onchange = function(e){
      state.settings.amountTolerance = Math.max(0, parseFloat(e.target.value) || 0);
      saveState(); renderAll();
    };
    document.getElementById('set-currency').onchange = function(e){
      state.settings.currency = e.target.value;
      saveState(); renderAll();
    };

    renderSettings();
